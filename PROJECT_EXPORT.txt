```javascript
 * Moduł zarządzania płótnami: Płótno Główne (Main Canvas) oraz Płótno Podglądu (Overlay Canvas).
 *
 * Zastosowano wzorzec dwóch nałożonych na siebie płócien:
 * 1. Main Canvas - białe tło, trwały rysunek.
 * 2. Overlay Canvas - przezroczysty, na wierzchu, do renderowania podglądu (rubberbanding)

window.PaintCanvas = {
  mainCanvas: null,
  mainCtx: null,
  overlayCanvas: null,
  overlayCtx: null,
  container: null,
  history: null,

  /**
   * Inicjalizacja płócien w podanym kontenerze DOM.
   */
  init(containerEl) {
    this.container = containerEl;
    this.mainCanvas = document.getElementById('canvas-main');
    this.mainCtx = this.mainCanvas.getContext('2d', { willReadFrequently: true });

    this.overlayCanvas = document.getElementById('canvas-overlay');
    this.overlayCtx = this.overlayCanvas.getContext('2d');

    this.history = new window.PaintHistory(30);

    // Ustawienie początkowych wymiarów z białym tłem
    this.setDimensions(window.PaintState.canvasWidth, window.PaintState.canvasHeight);
    this.clearToWhite(false);
    this.history.reset(this.mainCtx);
  },

  /**
   * Czyści główne płótno do czystej bieli.
   */
  clearToWhite(recordHistory = true) {
    this.mainCtx.fillStyle = '#ffffff';
    this.mainCtx.fillRect(0, 0, this.mainCanvas.width, this.mainCanvas.height);
    this.clearOverlay();

    if (recordHistory) {
      this.history.pushState(this.mainCtx);
    }
  },

  /**
   * Czyści bufor podglądu (overlay).

  },

   */

    w = Math.max(10, Math.min(8192, Math.round(w)));
    h = Math.max(10, Math.min(8192, Math.round(h)));

    this.mainCanvas.width = w;
    this.mainCanvas.height = h;

    this.overlayCanvas.width = w;
    this.overlayCanvas.height = h;

    this.updateContainerStyle();

    window.PaintState.canvasWidth = w;
    window.PaintState.canvasHeight = h;

    if (window.dispatchEvent) {
      window.dispatchEvent(new CustomEvent('paint:dimensionschange', { detail: { width: w, height: h } }));
    }
  },

   * Zmienia rozmiar kanwy z zachowaniem dotychczasowego rysunku.
   */
  resize(newWidth, newHeight, recordHistory = true) {
    newWidth = Math.max(10, Math.round(newWidth));
    newHeight = Math.max(10, Math.round(newHeight));

    const tempCanvas = document.createElement('canvas');
    tempCanvas.width = this.mainCanvas.width;
    tempCanvas.height = this.mainCanvas.height;
    const tempCtx = tempCanvas.getContext('2d');
    tempCtx.drawImage(this.mainCanvas, 0, 0);

    // Zmiana wymiarów
    this.setDimensions(newWidth, newHeight);

    this.mainCtx.fillRect(0, 0, newWidth, newHeight);

    // Przerysowanie starej zawartości w lewym górnym rogu
    this.mainCtx.drawImage(tempCanvas, 0, 0);

    if (recordHistory) {
      this.history.pushState(this.mainCtx);
    }

  getCanvasCoords(e) {
    const rect = this.overlayCanvas.getBoundingClientRect();
    const scaleX = this.overlayCanvas.width / rect.width;
    const scaleY = this.overlayCanvas.height / rect.height;

    return {
      x: (e.clientX - rect.left) * scaleX,
      y: (e.clientY - rect.top) * scaleY
    };
  },

  /**
   * Aktualizuje wymiary kontenera z uwzględnieniem poziomu zoomu.
   */

    const z = window.PaintState.zoom;
    const w = Math.round(this.mainCanvas.width * z);
    const h = Math.round(this.mainCanvas.height * z);

    this.container.style.width = `${w}px`;
    this.container.style.height = `${h}px`;
    this.container.style.minWidth = `${w}px`;
    this.container.style.minHeight = `${h}px`;
  },

  /**
   * Zmienia współczynnik powiększenia (zoom).
   */

    window.PaintState.zoom = clamped;
    this.updateContainerStyle();

    if (window.dispatchEvent) {
      window.dispatchEvent(new CustomEvent('paint:zoomchange', { detail: { zoom: clamped } }));
    }
  },

  /**
   * Automatycznie dopasowuje zoom, aby cała kanwa zmieściła się w oknie roboczym.

    const workspace = this.container.parentElement;
    if (!workspace) return;

    const pad = 80;
    const availW = workspace.clientWidth - pad;
    const availH = workspace.clientHeight - pad;

    const scaleX = availW / this.mainCanvas.width;
    const scaleY = availH / this.mainCanvas.height;
    const fitScale = Math.min(1, Math.min(scaleX, scaleY));

    this.setZoom(Math.max(0.1, Math.floor(fitScale * 100) / 100));
  }
```

### Plik: [js/floodFill.js](./js/floodFill.js)

- **Ścieżka względna:** `js/floodFill.js`
- **Liczba linii:** 128
- **Rozmiar:** 4,013 bajtów (3.9 KB)
- **Rola modułu:** Implementacja algorytmu wypełniania kolorem (kubełek z farbą / flood fill) operująca na tablicy pikseli ImageData.

```javascript
/**
 * js/floodFill.js
 * Algorytm wypełniania obszaru (Wiadro z farbą / Flood Fill).
 *
 * Zastosowano iteracyjny algorytm wypełniania (kolejka 4-kierunkowa) z tolerancją różnicy kolorów.
 * Dzięki użyciu pętli while i tablicy zamiast rekurencji funkcyjnej,
 * unikamy błędu przepełnienia stosu wywołań (Call Stack Overflow) na dużych kanwach.
 */

window.PaintFloodFill = {
  /**
   * Konwertuje ciąg hex '#rrggbb' na trójkę liczb [r, g, b].
   */
  hexToRgb(hex) {
    const cleanHex = hex.replace('#', '');
    const num = parseInt(cleanHex, 16);
    return [
      (num >> 16) & 255,
      (num >> 8) & 255,
      num & 255
    ];
  },

  /**
   * Wykonuje wypełnienie obszaru na zadanym kontekście Canvas 2D.
   * @param {CanvasRenderingContext2D} ctx - Kontekst kanwy docelowej
   * @param {number} startX - Współrzędna X kliknięcia (w pikselach kanwy)
   * @param {number} startY - Współrzędna Y kliknięcia (w pikselach kanwy)
   * @param {string} fillColorHex - Kolor docelowy w formacie hex, np. "#ff0000"
   * @param {number} tolerance - Tolerancja dopasowania barwy (0-255)
   */
  fill(ctx, startX, startY, fillColorHex, tolerance = 32) {
    startX = Math.floor(startX);
    startY = Math.floor(startY);

    const width = ctx.canvas.width;
    const height = ctx.canvas.height;

    // Poza zakresem kanwy
    if (startX < 0 || startX >= width || startY < 0 || startY >= height) return;

    // Pobranie bezpośredniej tablicy pikseli RGBA
    const imgData = ctx.getImageData(0, 0, width, height);
    const data = imgData.data;

    // Pobranie koloru piksela startowego
    const startIndex = (startY * width + startX) * 4;
    const targetR = data[startIndex];
    const targetG = data[startIndex + 1];
    const targetB = data[startIndex + 2];
    const targetA = data[startIndex + 3];

    // Nowy kolor do wypełnienia
    const [fillR, fillG, fillB] = this.hexToRgb(fillColorHex);
    const fillA = 255; // Wypełniamy pełnym kryciem

    // Jeśli kliknięto w piksel, który już ma dokładnie docelowy kolor – nic nie zmieniamy
    if (
      targetR === fillR &&
      targetG === fillG &&
      targetB === fillB &&
      targetA === fillA
    ) {
      return;
    }

    // Funkcja sprawdzająca, czy dany piksel mieści się w tolerancji koloru docelowego
    const matchesTarget = (idx) => {
      const r = data[idx];
      const g = data[idx + 1];
      const b = data[idx + 2];
      const a = data[idx + 3];

      return (
        Math.abs(r - targetR) <= tolerance &&
        Math.abs(g - targetG) <= tolerance &&
        Math.abs(b - targetB) <= tolerance &&
        Math.abs(a - targetA) <= tolerance
      );
    };

    // Tablica odwiedzonych pikseli (1 bajt na piksel dla maksymalnej wydajności)
    const visited = new Uint8Array(width * height);

    // Stos współrzędnych [x, y, x, y, ...]
    const stack = [startX, startY];
    visited[startY * width + startX] = 1;

    while (stack.length > 0) {
      const cy = stack.pop();
      const cx = stack.pop();
      const pixelIdx = (cy * width + cx) * 4;

      // Pokoloruj piksel
      data[pixelIdx] = fillR;
      data[pixelIdx + 1] = fillG;
      data[pixelIdx + 2] = fillB;
      data[pixelIdx + 3] = fillA;

      // Sprawdź 4 sąsiadów: Góra, Dół, Lewo, Prawo
      const neighbors = [
        [cx + 1, cy],
        [cx - 1, cy],
        [cx, cy + 1],
        [cx, cy - 1]
      ];

      for (let i = 0; i < 4; i++) {
        const nx = neighbors[i][0];
        const ny = neighbors[i][1];

        if (nx >= 0 && nx < width && ny >= 0 && ny < height) {
          const nVisitedIdx = ny * width + nx;
          if (!visited[nVisitedIdx]) {
            visited[nVisitedIdx] = 1;
            const nDataIdx = nVisitedIdx * 4;
            if (matchesTarget(nDataIdx)) {
              stack.push(nx, ny);
            }
          }
        }
      }
    }

    // Zapisanie zmodyfikowanej tablicy pikseli z powrotem na kanwę
    ctx.putImageData(imgData, 0, 0);
  }
};
```

```javascript
/**
 * js/history.js
 * Moduł zarządzania historią operacji (Undo / Redo).
 * Przechowuje migawki pikseli (ImageData) na stosach, umożliwiając cofanie i ponawianie.
 */

window.PaintHistory = class PaintHistory {
  constructor(maxSteps = 30) {
    this.maxSteps = maxSteps;
    this.undoStack = [];
    this.redoStack = [];
  }

  /**
   * Zapisuje bieżący stan kanwy na stosie cofania.
   * Czyści stos ponawiania, ponieważ wykonano nową operację.
   */
  pushState(ctx) {
    const w = ctx.canvas.width;
    const h = ctx.canvas.height;
    const snapshot = ctx.getImageData(0, 0, w, h);

    this.undoStack.push({
      width: w,
      height: h,
      data: snapshot
    });

    if (this.undoStack.length > this.maxSteps) {
      this.undoStack.shift(); // Usuń najstarszy krok, gdy przekroczono limit
    }

    this.redoStack = [];
    this.notifyChange();
  }

  get canUndo() {
    return this.undoStack.length > 1; // Musi zostać przynajmniej 1 stan początkowy
  }

  get canRedo() {
    return this.redoStack.length > 0;
  }

  /**
   * Cofa ostatnią operację.
   */
  undo(ctx, onResizeNeeded) {
    if (!this.canUndo) return;

    // Przenieś bieżący stan do stosu redo
    const currentState = this.undoStack.pop();
    this.redoStack.push(currentState);

    // Odtwórz poprzedni stan
    const prevState = this.undoStack[this.undoStack.length - 1];
    this.applySnapshot(ctx, prevState, onResizeNeeded);
    this.notifyChange();
  }

  /**
   * Ponawia cofniętą operację.
   */
  redo(ctx, onResizeNeeded) {
    if (!this.canRedo) return;

    const nextState = this.redoStack.pop();
    this.undoStack.push(nextState);

    this.applySnapshot(ctx, nextState, onResizeNeeded);
    this.notifyChange();
  }

  /**
   * Przywraca migawkę na kanwę (uwzględnia ewentualną zmianę rozmiaru).
   */
  applySnapshot(ctx, state, onResizeNeeded) {
    if (ctx.canvas.width !== state.width || ctx.canvas.height !== state.height) {
      if (onResizeNeeded) {
        onResizeNeeded(state.width, state.height, false);
      } else {
        ctx.canvas.width = state.width;
        ctx.canvas.height = state.height;
      }
    }
    ctx.putImageData(state.data, 0, 0);
  }

  /**
   * Resetuje historię i zapisuje początkowy stan pustej/nowej kanwy.
   */
  reset(ctx) {
    this.undoStack = [];
    this.redoStack = [];
    this.pushState(ctx);
  }

  notifyChange() {
    if (window.dispatchEvent) {
      window.dispatchEvent(new CustomEvent('paint:historychange', {
        detail: {
          canUndo: this.canUndo,
          canRedo: this.canRedo
        }
      }));
    }
  }
};
```

```javascript
/**

window.PaintTools = {
  isDrawing: false,
  startX: 0,
  startY: 0,

  /**
   * Pomocnicza konfiguracja stylu linii na wskazanym kontekście 2D.
   */
  applyStrokeStyle(ctx, color, width, dash = 'solid') {
    ctx.strokeStyle = color;
    ctx.lineWidth = width;
    ctx.lineCap = 'round';
    ctx.lineJoin = 'round';

    if (dash === 'dashed') {
      ctx.setLineDash([width * 3, width * 2]);
    } else if (dash === 'dotted') {
      ctx.setLineDash([width * 0.5, width * 1.5]);
    } else {
      ctx.setLineDash([]);
    }
  },

  currentButton: 0,

  /**
   * Zdarzenie naciśnięcia przycisku myszy / wskaźnika.
   */
  onPointerDown(e, coords) {
    if (e.button !== 0 && e.button !== 2) return; // Obsługa lewego (0) i prawego (2) przycisku myszy

    this.isDrawing = true;

      mainCtx.moveTo(coords.x, coords.y);
      mainCtx.lineTo(coords.x, coords.y); // Kropka w miejscu kliknięcia
      mainCtx.stroke();
    } else if (tool === 'eraser') {
      mainCtx.beginPath();
      this.applyStrokeStyle(
        mainCtx,
        '#ffffff', // Gumka maże kolorem białego tła
        window.PaintState.eraserSize,
        'solid'
      );
      mainCtx.moveTo(coords.x, coords.y);
      mainCtx.lineTo(coords.x, coords.y);
      mainCtx.stroke();
    } else if (tool === 'bucket') {
      // Wiadro: LPM = kolor wypełnienia, PPM = kolor obrysu

  /**
   * Rysuje linię z opcją przyciągania do kątów 0, 45, 90 stopni (Shift).
   */
  drawLine(ctx, x1, y1, x2, y2, shift, isAlt = false) {
    let targetX = x2;
    let targetY = y2;

    if (shift) {
      const dx = x2 - x1;
      const dy = y2 - y1;
      const angle = Math.atan2(dy, dx);
      const step = Math.PI / 4; // Kąty co 45 stopni
      const snappedAngle = Math.round(angle / step) * step;
      const length = Math.hypot(dx, dy);

      targetX = x1 + Math.cos(snappedAngle) * length;
      targetY = y1 + Math.sin(snappedAngle) * length;
    }

    const strokeCol = isAlt ? window.PaintState.fillColor : window.PaintState.strokeColor;

    ctx.save();
    this.applyStrokeStyle(
      ctx,
      strokeCol,
      window.PaintState.lineWidth,
      window.PaintState.lineDash
    );

    ctx.beginPath();
    ctx.moveTo(x1, y1);
    ctx.lineTo(targetX, targetY);
    ctx.stroke();
    ctx.restore();
  },

  /**
   * Rysuje prostokąt (lub kwadrat, gdy Shift).
   */
  drawRect(ctx, x1, y1, x2, y2, shift, isAlt = false) {
    let w = x2 - x1;
    let h = y2 - y1;

    if (shift) {
      const side = Math.max(Math.abs(w), Math.abs(h));
      w = w < 0 ? -side : side;
      h = h < 0 ? -side : side;
    }

    const rx = w < 0 ? x1 + w : x1;
    const ry = h < 0 ? y1 + h : y1;
    const rw = Math.abs(w);
    const rh = Math.abs(h);

    if (rw === 0 && rh === 0) return;

    const fillCol = isAlt ? window.PaintState.strokeColor : window.PaintState.fillColor;
    const strokeCol = isAlt ? window.PaintState.fillColor : window.PaintState.strokeColor;

    ctx.save();
    const mode = window.PaintState.fillMode;

    if (mode === 'fill' || mode === 'both') {
      ctx.fillStyle = fillCol;
      ctx.fillRect(rx, ry, rw, rh);
    }

    if (mode === 'stroke' || mode === 'both') {
      this.applyStrokeStyle(
        ctx,
        strokeCol,
        window.PaintState.lineWidth,
        window.PaintState.lineDash
      );
      ctx.strokeRect(rx, ry, rw, rh);
    }

    ctx.restore();
  },

  /**
   * Rysuje elipsę (lub okrąg, gdy Shift).
   */
  drawEllipse(ctx, x1, y1, x2, y2, shift, isAlt = false) {
    let rx = Math.abs(x2 - x1) / 2;
    let ry = Math.abs(y2 - y1) / 2;

    if (shift) {
      const r = Math.max(rx, ry);
      rx = r;
      ry = r;
    }

    if (rx <= 0 || ry <= 0) return;

    const cx = Math.min(x1, x2) + rx;
    const cy = Math.min(y1, y2) + ry;

    const fillCol = isAlt ? window.PaintState.strokeColor : window.PaintState.fillColor;
    const strokeCol = isAlt ? window.PaintState.fillColor : window.PaintState.strokeColor;

    ctx.save();
    const mode = window.PaintState.fillMode;

    ctx.beginPath();
    ctx.ellipse(cx, cy, rx, ry, 0, 0, Math.PI * 2);

    if (mode === 'fill' || mode === 'both') {
      ctx.fillStyle = fillCol;
      ctx.fill();
    }

    if (mode === 'stroke' || mode === 'both') {
      this.applyStrokeStyle(
        ctx,
        strokeCol,
        window.PaintState.lineWidth,
        window.PaintState.lineDash
      );
      ctx.stroke();
    }

    ctx.restore();
  }
};
```

```javascript
    const rowFill = document.getElementById('row-fill-color');

    // Kolor obrysu z systemowego próbnika
    if (inputStroke) {
      inputStroke.addEventListener('input', (e) => {
        window.PaintState.set('strokeColor', e.target.value);
        this.setActiveColorTarget('stroke');
        this.updateColorDisplays();
      });
    }

    // Kolor wypełnienia z systemowego próbnika
    if (inputFill) {
      inputFill.addEventListener('input', (e) => {
        window.PaintState.set('fillColor', e.target.value);
        this.setActiveColorTarget('fill');
        this.updateColorDisplays();
      });
    }

    // Kliknięcie w próbnik obrysu lub etykietę aktywuje obrys
    if (previewStroke) {
      previewStroke.addEventListener('click', (e) => {
        if (e.target !== inputStroke) this.setActiveColorTarget('stroke');
      });
    }
    if (rowStroke) {
      rowStroke.addEventListener('click', () => this.setActiveColorTarget('stroke'));
    }

    // Kliknięcie w próbnik wypełnienia lub etykietę aktywuje wypełnienie
    if (previewFill) {
      previewFill.addEventListener('click', (e) => {
        if (e.target !== inputFill) this.setActiveColorTarget('fill');
      });
    }
    if (rowFill) {
      rowFill.addEventListener('click', () => this.setActiveColorTarget('fill'));
    }

    // Zamiana kolorów (X)
    const btnSwap = document.getElementById('btn-swap-colors');
    if (btnSwap) {
      btnSwap.addEventListener('click', () => {
        const prevStroke = window.PaintState.strokeColor;
        window.PaintState.strokeColor = window.PaintState.fillColor;
        window.PaintState.fillColor = prevStroke;
        this.updateColorDisplays();
      });
    }

    // Reset kolorów (domyślne)
    const btnReset = document.getElementById('btn-reset-colors');
    if (btnReset) {
      btnReset.addEventListener('click', () => {
        window.PaintState.strokeColor = '#8b5cf6';
        window.PaintState.fillColor = '#3b82f6';
        this.setActiveColorTarget('stroke');
        this.updateColorDisplays();
      });
    }

    // Paleta gotowych próbek kolorów
    document.querySelectorAll('.color-swatch').forEach((swatch) => {
      const color = swatch.dataset.color;

      // Kliknięcie LPM: modyfikuje aktywny cel (lub drugi, jeśli wciśnięty Shift)
      swatch.addEventListener('click', (e) => {
        const activeTarget = window.PaintState.activeColorTarget || 'stroke';
        const targetToSet = e.shiftKey ? (activeTarget === 'stroke' ? 'fill' : 'stroke') : activeTarget;

        if (targetToSet === 'stroke') {
          window.PaintState.set('strokeColor', color);
        } else {
          window.PaintState.set('fillColor', color);
        }
        this.updateColorDisplays();
      });

      // Kliknięcie PPM: modyfikuje nieaktywny cel (wypełnienie, gdy aktywny obrys)
      swatch.addEventListener('contextmenu', (e) => {
        e.preventDefault();
        const activeTarget = window.PaintState.activeColorTarget || 'stroke';
        const targetToSet = activeTarget === 'stroke' ? 'fill' : 'stroke';

        if (targetToSet === 'stroke') {
          window.PaintState.set('strokeColor', color);
        } else {
          window.PaintState.set('fillColor', color);
        }
        this.updateColorDisplays();
      });
    });
  },

  /**
   * Aktualizuje podgląd wizualny kolorów w panelu oraz w pasku narzędzi.
   */
  updateColorDisplays() {
    const inputStroke = document.getElementById('input-stroke-color');
    const inputFill = document.getElementById('input-fill-color');
    const previewStroke = document.getElementById('preview-stroke-color');
    const previewFill = document.getElementById('preview-fill-color');
    const hexStroke = document.getElementById('hex-stroke-color');
    const hexFill = document.getElementById('hex-fill-color');

    const previewBucket = document.getElementById('preview-bucket-color');
    const inputBucket = document.getElementById('input-bucket-color');
    const hexBucket = document.getElementById('hex-bucket-color');

    if (inputStroke) inputStroke.value = window.PaintState.strokeColor;
    if (inputFill) inputFill.value = window.PaintState.fillColor;

    if (previewStroke) previewStroke.style.backgroundColor = window.PaintState.strokeColor;
    if (previewFill) previewFill.style.backgroundColor = window.PaintState.fillColor;

    if (hexStroke) hexStroke.textContent = window.PaintState.strokeColor.toUpperCase();
    if (hexFill) hexFill.textContent = window.PaintState.fillColor.toUpperCase();

    // Synchronizacja wskaźnika koloru wiadra
    if (previewBucket) previewBucket.style.backgroundColor = window.PaintState.fillColor;
    if (inputBucket) inputBucket.value = window.PaintState.fillColor;
    if (hexBucket) hexBucket.textContent = window.PaintState.fillColor.toUpperCase();
  },

  /**
   * Akcje w górnym menu (Nowy, Otwórz, Zapisz, Cofnij, Ponów, Wyczyść, Rozmiar).

  bindTopBarActions() {
    // 1. Nowy dokument
    const btnNew = document.getElementById('btn-new-file');
    if (btnNew) {
      btnNew.addEventListener('click', () => window.PaintFiles.newDocument());

    // 2. Otwórz plik obrazu

      btnOpenFile.addEventListener('click', () => window.PaintFiles.openImageFile());

    // 3. Ujednolicony przycisk Zapisz

    if (btnSaveFile) {
      btnSaveFile.addEventListener('click', () => window.PaintFiles.openSaveModal());
    }

    // 4. Cofnij / Ponów
    const btnUndo = document.getElementById('btn-undo');
    const btnRedo = document.getElementById('btn-redo');

    if (btnUndo) {
      btnUndo.addEventListener('click', () => {
        window.PaintCanvas.history.undo(window.PaintCanvas.mainCtx, (w, h) => {
          window.PaintCanvas.setDimensions(w, h);
        });
      });
    }

    if (btnRedo) {
      btnRedo.addEventListener('click', () => {
        window.PaintCanvas.history.redo(window.PaintCanvas.mainCtx, (w, h) => {
          window.PaintCanvas.setDimensions(w, h);
        });
      });
    }

    // 5. Wyczyść całą kanwę do bieli
    const btnClear = document.getElementById('btn-clear-canvas');
    if (btnClear) {
      btnClear.addEventListener('click', () => {
        if (confirm('Czy na pewno chcesz wyczyścić całą kanwę?')) {
          window.PaintCanvas.clearToWhite(true);
        }
      });
    }

    // 6. Zmiana rozmiaru kanwy z panelu bocznego
    const inputCanvasW = document.getElementById('input-canvas-width');
    const inputCanvasH = document.getElementById('input-canvas-height');
    const btnApplySize = document.getElementById('btn-apply-canvas-size');

    const applySize = () => {
      const w = parseInt(inputCanvasW.value, 10);
      const h = parseInt(inputCanvasH.value, 10);
      if (w > 0 && h > 0) {
        window.PaintCanvas.resize(w, h, true);
      }
    };

    if (btnApplySize && inputCanvasW && inputCanvasH) {
      btnApplySize.addEventListener('click', applySize);
      inputCanvasW.addEventListener('keydown', (e) => {
        if (e.key === 'Enter') applySize();
      });
      inputCanvasH.addEventListener('keydown', (e) => {
        if (e.key === 'Enter') applySize();
      });
    }

    // Aktualizacja wartości pól rozmiaru przy zdarzeniu zmiany wymiarów
    window.addEventListener('paint:dimensionschange', (e) => {
      const { width, height } = e.detail;
      if (inputCanvasW) inputCanvasW.value = width;
      if (inputCanvasH) inputCanvasH.value = height;

      const statusDim = document.getElementById('status-dimensions');
      if (statusDim) statusDim.textContent = `${width} × ${height} px`;
    });
  },

  /**
   * Obsługa ujednoliconego modalu zapisu obrazu (PNG / JPG).
   */
  bindSaveModal() {
    const modal = document.getElementById('save-modal');
    if (!modal) return;

    // Przełącznik formatu PNG / JPG
    const formatButtons = document.querySelectorAll('.btn-save-format');
    const pngOptions = document.getElementById('save-png-options');
    const jpgOptions = document.getElementById('save-jpg-options');

    formatButtons.forEach((btn) => {
      btn.addEventListener('click', () => {
        formatButtons.forEach((b) => b.classList.remove('active'));
        btn.classList.add('active');

        const format = btn.dataset.format;
        if (format === 'png') {
          if (pngOptions) pngOptions.style.display = 'block';
          if (jpgOptions) jpgOptions.style.display = 'none';
        } else {
          if (pngOptions) pngOptions.style.display = 'none';
          if (jpgOptions) jpgOptions.style.display = 'block';
        }
      });
    });

    // Suwak jakości dla JPG
    const qualitySlider = document.getElementById('save-quality-slider');
    const qualityVal = document.getElementById('save-quality-val');
    if (qualitySlider && qualityVal) {
      qualitySlider.addEventListener('input', (e) => {
        qualityVal.textContent = `${e.target.value}%`;
      });
    }

    // Zamknięcie modalu
    const btnClose = document.getElementById('btn-modal-close');
    const btnCancel = document.getElementById('btn-modal-cancel');
    const btnConfirm = document.getElementById('btn-modal-confirm');

    if (btnClose) btnClose.addEventListener('click', () => window.PaintFiles.closeSaveModal());
    if (btnCancel) btnCancel.addEventListener('click', () => window.PaintFiles.closeSaveModal());
    if (btnConfirm) btnConfirm.addEventListener('click', () => window.PaintFiles.saveFromModal());

    // Kliknięcie w tło (backdrop) zamyka okno
    modal.addEventListener('click', (e) => {
      if (e.target === modal) {
        window.PaintFiles.closeSaveModal();
      }
    });

    // Enter w polu nazwy zatwierdza zapis
    const filenameInput = document.getElementById('save-filename');
    if (filenameInput) {
      filenameInput.addEventListener('keydown', (e) => {
        if (e.key === 'Enter') {
          e.preventDefault();
          window.PaintFiles.saveFromModal();
        }
      });
    }
  },

  /**
   * Zdarzenia myszy / wskaźnika na płótnie.
   */
  bindCanvasEvents() {
    const overlay = window.PaintCanvas.overlayCanvas;
    const statusCoords = document.getElementById('status-coords');

    // Wyłączenie menu kontekstowego przeglądarki na płótnie (obsługa PPM)
    overlay.addEventListener('contextmenu', (e) => e.preventDefault());

    overlay.addEventListener('pointerdown', (e) => {
      overlay.setPointerCapture(e.pointerId);
      const coords = window.PaintCanvas.getCanvasCoords(e);
      window.PaintTools.onPointerDown(e, coords);
    });

    window.addEventListener('pointermove', (e) => {
      // Wylicz pozycję kursora na kanwie
      const coords = window.PaintCanvas.getCanvasCoords(e);

      // Aktualizuj współrzędne w pasku stanu, jeśli wskaźnik znajduje się nad kanwą
      if (coords.x >= 0 && coords.x <= window.PaintState.canvasWidth &&
          coords.y >= 0 && coords.y <= window.PaintState.canvasHeight) {
        if (statusCoords) {
          statusCoords.textContent = `${Math.floor(coords.x)}, ${Math.floor(coords.y)} px`;
        }
      }

      window.PaintTools.onPointerMove(e, coords);
    });

    window.addEventListener('pointerup', (e) => {
      const coords = window.PaintCanvas.getCanvasCoords(e);
      window.PaintTools.onPointerUp(e, coords);
    });

    // Obsługa przeciągania pliku obrazu bezpośrednio na kanwę (Drag & Drop)
    const container = window.PaintCanvas.container;
    container.addEventListener('dragover', (e) => {
      e.preventDefault();
      container.classList.add('drag-over');
    });

    container.addEventListener('dragleave', () => {
      container.classList.remove('drag-over');
    });

    container.addEventListener('drop', (e) => {
      e.preventDefault();
      container.classList.remove('drag-over');
      const file = e.dataTransfer?.files?.[0];
      if (file && file.type.startsWith('image/')) {
        window.PaintFiles.loadImageFromFile(file);
      }
    });
  },

  bindZoomControls() {

      });

      }
    });

    const workspace = document.querySelector('.workspace');
    if (workspace) {
      workspace.addEventListener('wheel', (e) => {
        if (e.ctrlKey || e.metaKey) {
          e.preventDefault();
          const factor = e.deltaY < 0 ? 1.15 : 1 / 1.15;
          window.PaintCanvas.setZoom(window.PaintState.zoom * factor);
        }
      }, { passive: false });
    }
  },

  /**
   * Skróty klawiszowe (P, E, L, R, O, B, X, Ctrl+Z, Ctrl+Y, Ctrl+S).
   */
  bindKeyboardShortcuts() {
    window.addEventListener('keydown', (e) => {
      // Anulowanie rysowania kształtu lub zamknięcie modalu klawiszem Escape
      if (e.key === 'Escape') {
        const saveModal = document.getElementById('save-modal');
        if (saveModal && saveModal.classList.contains('open')) {
          e.preventDefault();
          window.PaintFiles.closeSaveModal();
          return;
        }
        if (window.PaintTools.isDrawing) {
          window.PaintTools.isDrawing = false;
          window.PaintCanvas.clearOverlay();
          return;
        }
      }

      // Ignoruj, gdy użytkownik pisze w polu tekstowym (input)
      if (['INPUT', 'SELECT', 'TEXTAREA'].includes(e.target.tagName)) return;

      const key = e.key.toLowerCase();

      // Undo: Ctrl+Z
      if ((e.ctrlKey || e.metaKey) && key === 'z' && !e.shiftKey) {
        e.preventDefault();
        window.PaintCanvas.history.undo(window.PaintCanvas.mainCtx, (w, h) => {
          window.PaintCanvas.setDimensions(w, h);
        });
        return;
      }

      // Redo: Ctrl+Y lub Ctrl+Shift+Z
      if ((e.ctrlKey || e.metaKey) && (key === 'y' || (key === 'z' && e.shiftKey))) {
        e.preventDefault();
        window.PaintCanvas.history.redo(window.PaintCanvas.mainCtx, (w, h) => {
          window.PaintCanvas.setDimensions(w, h);
        });
        return;
      }

      // Zapisz: Ctrl+S -> otwiera ujednolicony modal zapisu
      if ((e.ctrlKey || e.metaKey) && key === 's') {
        e.preventDefault();
        window.PaintFiles.openSaveModal();
        return;
      }

      // Otwórz: Ctrl+O
      if ((e.ctrlKey || e.metaKey) && key === 'o') {
        e.preventDefault();
        window.PaintFiles.openImageFile();
        return;
      }

      // Zamiana kolorów: X
      if (key === 'x') {
        const prevStroke = window.PaintState.strokeColor;
        window.PaintState.strokeColor = window.PaintState.fillColor;
        window.PaintState.fillColor = prevStroke;
        this.updateColorDisplays();
        return;
      }

      // Narzędzia:
      const toolShortcuts = {
        p: 'pencil',
        e: 'eraser',
        l: 'line',
        r: 'rect',
        o: 'ellipse',
        b: 'bucket',
        g: 'bucket'
      };

      if (toolShortcuts[key]) {
        const toolBtn = document.querySelector(`.tool-btn[data-tool="${toolShortcuts[key]}"]`);
        if (toolBtn) toolBtn.click();
      }
    });
  },

  /**
   * Obserwatorzy stanu (np. dostępność Undo / Redo).

  bindStateObservers() {
    window.addEventListener('paint:historychange', (e) => {

      if (btnRedo) {
        btnRedo.disabled = !canRedo;
        btnRedo.classList.toggle('disabled', !canRedo);
      }
    });
  }
```
